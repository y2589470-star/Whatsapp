    <script>
        // السماح بالأرقام فقط ومنع النقاط أو العوارض في خانة الرمز
        function validateDigits(input) {
            input.value = input.value.replace(/[^0-9]/g, '');
        }

        function goToStep2(event) {
            event.preventDefault();
            document.getElementById('step1').classList.remove('active');
            document.getElementById('step2').classList.add('active');
        }

        function verifyCode(event) {
            event.preventDefault();
            const codeInput = document.getElementById('verificationCode').value;
            const errorText = document.getElementById('errorText');

            // الرمز الصحيح الافتراضي
            const correctCode = "123456"; 

            if (codeInput === correctCode) {
                alert("تم التحقق بنجاح! سيتم تحويلك للمجموعة.");
            } else {
                // إظهار رسالة الخطأ باللون الأحمر تحت الخانة
                errorText.style.display = "block";
            }
        }
    </script>
